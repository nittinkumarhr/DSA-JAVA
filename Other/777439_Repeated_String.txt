public static long repeatedString(String s, long n) {

     *  2. LONG_INTEGER n
     */
     *  1. STRING s
        long countInSingle = 0;
        long countInRemainder = 0;
        int stringLength = s.length();
        
        // Calculate full repetitions and the remaining characters
        long fullRepeats = n / stringLength;
        long remainderLength = n % stringLength;
        
        // Count 'a's in the full string and the remainder prefix in a single pass
        for (int i = 0; i < stringLength; i++) {
            if (s.charAt(i) == 'a') {
                countInSingle++;
                if (i < remainderLength) {
                    countInRemainder++;
                }
            }
        }
        
        // Return total count using long to prevent integer overflow
        return (countInSingle * fullRepeats) + countInRemainder;
    }
}
